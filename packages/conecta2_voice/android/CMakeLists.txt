cmake_minimum_required(VERSION 3.22.1)
project(conecta2_voice LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../native/whisper whisper-build)
add_library(conecta2_voice SHARED ../ios/conecta2_voice/Sources/VoiceBridge/voice_bridge.cpp)
target_link_libraries(conecta2_voice PRIVATE whisper)
target_compile_features(conecta2_voice PRIVATE cxx_std_17)
target_link_options(conecta2_voice PRIVATE "-Wl,-z,max-page-size=16384")
